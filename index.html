    <script>

        const CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vTAfSP2o2DjZ30zAwJLm3yQGIXzvtuz5k8Y5lkcpcfeuIcHWgAaGGv3169AAIT_typ-gAXvlNE7ka4A/pub?gid=0&single=true&output=csv";


        function switchTab(tabId) {

            ['matches', 'stats'].forEach(tab => {

                const section = document.getElementById('section-' + tab);

                const button = document.getElementById('tab-' + tab);


                if (section) section.classList.add('hidden');

                if (button) {

                    button.className = 'flex-1 py-2 text-xs font-bold rounded-lg transition-all text-gray-400 hover:text-white';

                }

            });


            const chosenSection = document.getElementById('section-' + tabId);

            const chosenButton = document.getElementById('tab-' + tabId);


            if (chosenSection) chosenSection.classList.remove('hidden');

            if (chosenButton) {

                chosenButton.className = 'flex-1 py-2 text-xs font-bold rounded-lg transition-all bg-blue-600 text-white shadow';

            }

        }


        function shareToWhatsApp() {

            const text = "🎱 Oxford Blues Pool Update: Check out our latest match reports and player win tallies on our live team dashboard.";

            window.open("https://api.whatsapp.com/send?text=" + encodeURIComponent(text), "_blank");

        }


        function getValue(row, heading) {

            const key = Object.keys(row).find(

                item => item.trim().toLowerCase() === heading.trim().toLowerCase()

            );

            return key ? String(row[key] ?? '').trim() : '';

        }


        function escapeHtml(value) {

            return String(value ?? '').replace(/[&<>"']/g, char => ({

                '&': '&amp;',

                '<': '&lt;',

                '>': '&gt;',

                '"': '&quot;',

                "'": '&#39;'

            })[char]);

        }


        function formatWinners(value) {

            return String(value || '')

                .split(/[,;]/)

                .map(name => name.trim())

                .filter(Boolean);

        }


        function loadDashboard() {

            Papa.parse(CSV_URL, {

                download: true,

                header: true,

                skipEmptyLines: true,

                complete: function(result) {

                    const matches = result.data.filter(row => {

                        return getValue(row, 'Opponent') || getValue(row, 'Blues Score');

                    });


                    const matchCountEl = document.getElementById('match-count');

                    if (matchCountEl) {

                        matchCountEl.textContent = `${matches.length} Match${matches.length === 1 ? '' : 'es'} Played`;

                    }


                    const matchListEl = document.getElementById('match-list');

                    const playerStatsEl = document.getElementById('player-stats');


                    if (!matches.length) {

                        if (matchListEl) {

                            matchListEl.textContent = 'No match results found. Check that your published sheet has rows.';

                        }

                        if (playerStatsEl) {

                            playerStatsEl.textContent = 'No player stats yet.';

                        }

                        return;

                    }


                    const tallies = {};


                    if (matchListEl) {

                        matchListEl.innerHTML = matches.map((row, index) => {

                            const week = getValue(row, 'Game Week');

                            const opponent = getValue(row, 'Opponent') || 'Opponent';

                            const bluesScore = getValue(row, 'Blues Score');

                            const opponentScore = getValue(row, 'Opponent Score');

                            const venue = getValue(row, 'Venue');

                            const report = getValue(row, 'Report');

                            const winners = formatWinners(getValue(row, 'Singles Winners'));


                            winners.forEach(name => {

                                tallies[name] = (tallies[name] || 0) + 1;

                            });


                            const blues = Number(bluesScore);

                            const opp = Number(opponentScore);

                            const isDraw = Number.isFinite(blues) && Number.isFinite(opp) && blues === opp;

                            const isWin = Number.isFinite(blues) && Number.isFinite(opp) && blues > opp;

                            const scoreColor = isDraw ? 'text-yellow-400' : isWin ? 'text-emerald-400' : 'text-red-400';

                            const resultLabel = isDraw ? 'Draw' : isWin ? 'Win' : 'Loss';

                            const borderColor = isDraw ? 'border-l-yellow-500' : isWin ? 'border-l-emerald-500' : 'border-l-red-500';


                            return `

                                <article class="match-card glass-card rounded-2xl p-4 shadow-xl border-l-4 ${borderColor}">

                                    <div class="flex justify-between items-start mb-2">

                                        <div>

                                            <span class="text-[10px] uppercase tracking-wider font-bold text-blue-300 bg-blue-500/10 px-2 py-0.5 rounded">

                                                ${index === 0 ? 'Latest Match' : escapeHtml(resultLabel)}

                                            </span>

                                            <h3 class="text-lg font-bold text-white mt-1">

                                                Oxford Blues <span class="${scoreColor}">${escapeHtml(bluesScore)}-${escapeHtml(opponentScore)}</span> ${escapeHtml(opponent)}

                                            </h3>

                                            <p class="text-xs text-gray-400 mt-1">

                                                ${week ? `Game Week ${escapeHtml(week)}` : ''}

                                                ${venue ? ` · ${escapeHtml(venue)}` : ''}

                                            </p>

                                        </div>

                                        <span class="text-xl">🎱</span>

                                    </div>

                                    <div class="space-y-3 mt-3 text-xs text-gray-300 border-t border-gray-800/80 pt-3">

                                        ${report ? `

                                            <div>

                                                <span class="font-semibold text-blue-300 block mb-1">Captain's Log:</span>

                                                <p class="leading-relaxed text-gray-300 whitespace-pre-line">${escapeHtml(report)}</p>

                                            </div>

                                        ` : ''}


                                        ${winners.length ? `

                                            <div class="bg-gray-900/50 p-2.5 rounded-xl border border-gray-800">

                                                <span class="font-semibold text-emerald-400 block mb-1">Singles Match Winners:</span>

                                                <div class="flex flex-wrap gap-1.5">

                                                    ${winners.map(name => `

                                                        <span class="bg-emerald-950 text-emerald-300 border border-emerald-800/60 px-2 py-0.5 rounded-md font-medium">

                                                            ${escapeHtml(name)}

                                                        </span>

                                                    `).join('')}

                                                </div>

                                            </div>

                                        ` : ''}

                                    </div>

                                </article>

                            `;

                        }).join('');

                    }


                    const entries = Object.entries(tallies).sort((a, b) => b[1] - a[1]);


                    if (playerStatsEl) {

                        playerStatsEl.innerHTML = entries.length

                            ? entries.map(([name, wins], index) => `

                                <div class="flex items-center justify-between py-2 ${index < entries.length - 1 ? 'border-b border-gray-800' : ''}">

                                    <span class="font-semibold text-sm text-white flex items-center gap-2">

                                        <span class="w-2 h-2 rounded-full bg-blue-500"></span>

                                        ${escapeHtml(name)}

                                    </span>

                                    <span class="bg-blue-600/20 text-blue-400 font-bold px-3 py-1 rounded-lg text-xs border border-blue-500/30">

                                        ${wins} ${wins === 1 ? 'Win' : 'Wins'}

                                    </span>

                                </div>

                            `).join('')

                            : 'No singles winners listed yet.';

                    }

                },

                error: function() {

                    const matchListEl = document.getElementById('match-list');

                    const playerStatsEl = document.getElementById('player-stats');


                    if (matchListEl) {

                        matchListEl.textContent = 'Could not load the spreadsheet. Check that the tab is published as CSV.';

                    }

                    if (playerStatsEl) {

                        playerStatsEl.textContent = 'Could not load player stats.';

                    }

                }

            });

        }


        loadDashboard();

    </script>
